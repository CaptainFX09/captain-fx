<script>
  // profit-split calculator
  const slider = document.getElementById('amtSlider');
  const amtOut = document.getElementById('amtOut');
  const outResult = document.getElementById('outResult');
  const outClient = document.getElementById('outClient');
  const outMgr = document.getElementById('outMgr');

  function fmt(n){
    const sign = n < 0 ? '-' : '';
    return sign + '$' + Math.abs(n).toFixed(2);
  }

  function update(){
    const val = parseInt(slider.value, 10);
    amtOut.textContent = val.toLocaleString();
    outResult.textContent = fmt(val);
    outClient.textContent = fmt(val * 0.6);
